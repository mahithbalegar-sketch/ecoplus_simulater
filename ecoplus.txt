# 2. SCIENCE ENGINE & SIMULATION MATHEMATICS
# ==========================================
@st.cache_data
def calculate_climate_projection(renewables_pct, forest_pct, carbon_capture_pct, methane_reduction_pct):
    """
    Simplified climate model parameterization inspired by IPCC AR6 SSP scenarios.
    Returns projected timeline from 2026 to 2056.
    """
    years = np.arange(2026, 2057)
    n_years = len(years)

    # Baseline SSP2-4.5 (Business as usual baseline)
    base_co2_ppm = np.linspace(424, 485, n_years)  # Current CO2 ~ 424 ppm
    base_temp_anomaly = np.linspace(1.25, 2.10, n_years)  # Deg C above pre-industrial
    base_sea_level = np.linspace(0, 18.5, n_years)  # cm increase from 2026

    # Calculate Action Mitigation Factors (0.0 to 1.0 impact scale)
    energy_impact = (renewables_pct / 100.0) * 0.35
    forest_impact = (forest_pct / 100.0) * 0.15
    ccs_impact = (carbon_capture_pct / 100.0) * 0.20
    methane_impact = (methane_reduction_pct / 100.0) * 0.30

    total_mitigation_score = energy_impact + forest_impact + ccs_impact + methane_impact

    # Calculate Policy-Adjusted Trajectories
    sim_co2_ppm = base_co2_ppm - (years - 2026) * (total_mitigation_score * 1.4)
    sim_temp_anomaly = base_temp_anomaly - (years - 2026) * (total_mitigation_score * 0.022)
    sim_sea_level = base_sea_level - (years - 2026) * (total_mitigation_score * 0.22)

    # Ensure physical lower bounds based on committed thermal inertia
    sim_temp_anomaly = np.maximum(sim_temp_anomaly, 1.1)

    df = pd.DataFrame({
        "Year": years,
        "Base_Temp_Anomaly": np.round(base_temp_anomaly, 3),
        "Sim_Temp_Anomaly": np.round(sim_temp_anomaly, 3),
        "Base_CO2_PPM": np.round(base_co2_ppm, 1),
        "Sim_CO2_PPM": np.round(sim_co2_ppm, 1),
        "Base_Sea_Level_CM": np.round(base_sea_level, 2),
        "Sim_Sea_Level_CM": np.round(sim_sea_level, 2)
    })

    return df, total_mitigation_score
